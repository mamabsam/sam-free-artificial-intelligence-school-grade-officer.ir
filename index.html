/* تبدیل دقیقه به "17:35" */
function minToTime(min){

}

/* ساخت ردیف‌های زمان — مرتب‌شده بر اساس نزدیک‌ترین به الان */
function buildTimesRowsHTML(list, nowMin){
if (!list || !list.length){
return '<div class="metro-empty">در این جهت قطاری ثبت نشده (مبدأ/مقصد)</div>';
}
// پیدا کردن ایندکس نزدیک‌ترین زمان آینده
let nextIdx = -1;
for (let i = 0; i < list.length; i++){
if (timeToMin(list[i]) >= nowMin){
nextIdx = i;
break;
}
}
// مرتب‌سازی: چند تا قبل + چند تا بعد (5 قبل + 6 بعد)
let rows = [];
const start = nextIdx < 0 ? list.length - 5 : Math.max(0, nextIdx - 3);
const end = nextIdx < 0 ? list.length : Math.min(list.length, nextIdx + 7);

for (let i = 0; i < list.length; i++){
const t = list[i];
const tMin = timeToMin(t);
const isPast = tMin < nowMin;
const isNext = (i === nextIdx);
if (i >= start && i < end){
const cls = isNext ? 'next' : (isPast ? 'past' : '');
const label = isNext ? '⏭️ قطار بعدی' : (isPast ? 'قطار قبلی' : 'قطار آینده');
let countdown = '';
if (isNext){
countdown = '<span class="metro-countdown">(' + (tMin - nowMin) + ' دقیقه دیگر)</span>';
} else if (!isPast){
countdown = '<span class="metro-countdown">(' + (tMin - nowMin) + ' دقیقه دیگر)</span>';
} else {
countdown = '<span class="metro-countdown">(رفت)</span>';
}
rows.push('<div class="metro-time-row ' + cls + '">' +
'<span class="metro-time-label">' + label + '</span>' +
'<span><span class="metro-time-val">' + t + '</span> ' + countdown + '</span>' +
'</div>');
}
}
if (rows.length === 0){
return '<div class="metro-empty">امروز قطاری در این ساعت باقی نمانده</div>';
}
return rows.join('');
}

function getScheduleAnswer(userText){
const t = (userText || '').toLowerCase();
const today = getToday();
const tomorrow = getTomorrow();
if (/(فردا|tomorrow)/i.test(t) && SCHEDULE[tomorrow]){
let reply = '📅 برنامه فردا (' + tomorrow + '):\n\n';
SCHEDULE[tomorrow].forEach(function(l, i){
reply += '📚 زنگ ' + (i + 1) + ': ' + l.subject + ' (' + l.teacher + ')\n';
});
reply += '\n⏰ ساعت ۶:۳۰ بیدار شو!';
return reply;
}
if (/(امروز|today)/i.test(t) && SCHEDULE[today]){
let reply = '📅 برنامه امروز (' + today + '):\n\n';
SCHEDULE[today].forEach(function(l, i){
reply += '📚 زنگ ' + (i + 1) + ': ' + l.subject + ' (' + l.teacher + ')\n';
});
return reply;
}
if (/(برنامه|زنگ|کلاس|درس)/i.test(t)){
let reply = '📅 برنامه هفتگی:\n\n';
for (const day in SCHEDULE){
reply += '• ' + day + ': ' + SCHEDULE[day].map(function(l){ return l.subject; }).join(' | ') + '\n\n';
}
return reply;
}
if (/(بیدار)/i.test(t)){
if (SCHEDULE[tomorrow]){
return '⏰ برای زنگ اول فردا ساعت ۶:۳۰ بیدار شو.\n📚 زنگ اول: ' + SCHEDULE[tomorrow][0].subject;
}
return 'فردا تعطیله! 😴';
}
return null;
}

function toggleMic(){
const SR = window.SpeechRecognition || window.webkitSpeechRecognition;
if (!SR){ toast('❌ مرورگر شما از ضبط صدا پشتیبانی نمی‌کند'); return; }
const btn = document.getElementById('micBtn');
if (isRecording){
if (recognition) recognition.stop();
return;
}
recognition = new SR();
recognition.lang = 'fa-IR';
recognition.continuous = false;
recognition.interimResults = true;
recognition.onstart = function(){
isRecording = true;
btn.classList.add('recording');
toast('🎤 در حال ضبط... صحبت کن');
};
recognition.onresult = function(e){
let final = '';
for (let i = e.resultIndex; i < e.results.length; i++){
if (e.results[i].isFinal) final += e.results[i][0].transcript;
}
if (final){
ci.value = (ci.value + ' ' + final).trim();
ci.dispatchEvent(new Event('input'));
}
};
recognition.onerror = function(e){
toast('❌ خطا در ضبط: ' + e.error);
stopMic();
};
recognition.onend = stopMic;
try{ recognition.start(); }catch(e){ toast('❌ خطا در شروع ضبط'); }
}
function stopMic(){
isRecording = false;
const btn = document.getElementById('micBtn');
if (btn) btn.classList.remove('recording');
}
